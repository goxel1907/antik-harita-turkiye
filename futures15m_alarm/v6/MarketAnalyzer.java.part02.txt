        if (x >= -0.15) return (int)Math.round(max * 0.45);
        return 0;
    }

    private static int takerPoints(Metrics m, boolean longSide, int max) {
        if (Double.isNaN(m.takerBuySellRatio)) return max / 3;
        double r = m.takerBuySellRatio;
        if (longSide) {
            if (r >= 1.18) return max;
            if (r >= 1.03) return (int)Math.round(max * 0.72);
            if (r >= 0.95) return (int)Math.round(max * 0.35);
        } else {
            if (r <= 0.85) return max;
            if (r <= 0.97) return (int)Math.round(max * 0.72);
            if (r <= 1.05) return (int)Math.round(max * 0.35);
        }
        return 0;
    }

    private static int momentumPoints(Frame f15, boolean longSide, int max) {
        double r = f15.rsi;
        if (longSide) {
            if (r >= 48 && r <= 68) return max;
            if (r >= 42 && r <= 75) return (int)Math.round(max * 0.65);
        } else {
            if (r >= 32 && r <= 55) return max;
            if (r >= 25 && r <= 62) return (int)Math.round(max * 0.65);
            if (r >= 70) return (int)Math.round(max * 0.60); // exhaustion helps resistance shorts only mildly
        }
        return 0;
    }

    private static int sentimentPoints(Metrics m, boolean longSide, int max) {
        int s = 0;
        if (!Double.isNaN(m.fundingRate)) {
            double f = m.fundingRate;
            if (longSide) {
                if (f < 0.0005) s += Math.max(1, max / 2);
                if (f < -0.0002) s += 1;
            } else {
                if (f > -0.0005) s += Math.max(1, max / 2);
                if (f > 0.0004) s += 1;
            }
        } else s += Math.max(1, max / 3);
        if (!Double.isNaN(m.globalLongShortRatio)) {
            double r = m.globalLongShortRatio;
            if (longSide && r < 2.0) s++;
            if (!longSide && r > 0.55) s++;
        }
        if (!Double.isNaN(m.topPositionLongShortRatio)) {
            double r = m.topPositionLongShortRatio;
            if (longSide && r >= 0.9) s++;
            if (!longSide && r <= 1.25) s++;
        }
        return Math.min(max, s);
    }

    private static double nearestSupport(double price, double atr, List<Bar> m15, List<Bar> h1, double ema1h20) {
        ArrayList<Double> c = new ArrayList<>();
        pivots(m15, false, c, 90);
        pivots(h1, false, c, 55);
        if (ema1h20 > 0 && ema1h20 < price) c.add(ema1h20);
        c.add(lowest(m15, 18));
        c.add(lowest(h1, 8));
        double best = -1;
        for (double x : c) if (x > 0 && x < price - 0.05 * atr && x > best) best = x;
        return best;
    }

    private static double nearestResistance(double price, double atr, List<Bar> m15, List<Bar> h1, double ema1h20) {
        ArrayList<Double> c = new ArrayList<>();
        pivots(m15, true, c, 90);
        pivots(h1, true, c, 55);
        if (ema1h20 > price) c.add(ema1h20);
        c.add(highest(m15, 18));
        c.add(highest(h1, 8));
        double best = Double.MAX_VALUE;
        for (double x : c) if (x > price + 0.05 * atr && x < best) best = x;
        return best == Double.MAX_VALUE ? -1 : best;
    }

    private static void pivots(List<Bar> bars, boolean highs, List<Double> out, int lookback) {
        int n = bars.size();
        int start = Math.max(2, n - lookback);
        for (int i = start; i < n - 2; i++) {
            double v = highs ? bars.get(i).high : bars.get(i).low;
            boolean p = true;
            for (int j = i - 2; j <= i + 2; j++) {
                if (j == i) continue;
                double x = highs ? bars.get(j).high : bars.get(j).low;
                if (highs ? x >= v : x <= v) { p = false; break; }
            }
            if (p) out.add(v);
        }
    }

    private static double highest(List<Bar> b, int n) {
        double x = -Double.MAX_VALUE;
        for (int i = Math.max(0, b.size() - n); i < b.size(); i++) x = Math.max(x, b.get(i).high);
        return x;
    }
    private static double lowest(List<Bar> b, int n) {
        double x = Double.MAX_VALUE;
        for (int i = Math.max(0, b.size() - n); i < b.size(); i++) x = Math.min(x, b.get(i).low);
        return x;
    }

    private static boolean bullishRejection(Bar b, Bar prev, double atr) {
        double body = Math.abs(b.close - b.open);
        double lower = Math.min(b.open, b.close) - b.low;
        double range = Math.max(b.high - b.low, 1e-12);
        return (b.close > b.open && b.close >= b.low + 0.58 * range && (lower >= body * 0.55 || b.close > prev.close)) ||
                (lower >= 0.32 * range && b.close > prev.close && range >= 0.55 * atr);
    }

    private static boolean bearishRejection(Bar b, Bar prev, double atr) {
        double body = Math.abs(b.close - b.open);
        double upper = b.high - Math.max(b.open, b.close);
        double range = Math.max(b.high - b.low, 1e-12);
        return (b.close < b.open && b.close <= b.low + 0.42 * range && (upper >= body * 0.55 || b.close < prev.close)) ||
                (upper >= 0.32 * range && b.close < prev.close && range >= 0.55 * atr);
    }

    private static double closeLocation(Bar b) { return (b.close - b.low) / Math.max(b.high - b.low, 1e-12); }
    private static boolean overlaps(double l1, double h1, double l2, double h2) { return h1 >= l2 && l1 <= h2; }
    private static double distanceToZone(double p, double l, double h) { if (p >= l && p <= h) return 0; return p < l ? l - p : p - h; }

    private static double targetLong(double entry, double risk, double resistance, double r) {
        double rr = entry + r * risk;
        if (resistance > entry + 0.90 * risk && resistance < entry + 1.90 * risk) return resistance;
        return rr;
    }
    private static double targetShort(double entry, double risk, double support, double r) {
        double rr = entry - r * risk;
        if (support < entry - 0.90 * risk && support > entry - 1.90 * risk) return support;
        return rr;
    }
    private static double rr(double e, double s, double t) { return (t - e) / Math.max(e - s, 1e-12); }
