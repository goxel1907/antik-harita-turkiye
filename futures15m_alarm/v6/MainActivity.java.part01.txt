    private void addSetup(LinearLayout card, String title, String trigger, String detail, boolean isLong) {
        String direction = isLong ? "🟢 LONG" : "🔴 SHORT";
        int color = isLong ? Color.rgb(74, 222, 128) : Color.rgb(248, 113, 113);
        TextView h = text(direction + " • " + title.replace("LONG ", "").replace("SHORT ", ""), 14, color, true);
        h.setPadding(0, dp(6), 0, dp(2));
        card.addView(h);
        String d = detail == null || detail.trim().isEmpty()
                ? "STOP/TP henüz yok — detaylı plan koduyla güncelle"
                : detail;
        TextView body = text(trigger + "\n" + d, 14, Color.rgb(226, 232, 240), false);
        body.setLineSpacing(0, 1.14f);
        card.addView(body);
    }

    private String autoReason(AutoSnapshot s, String which) {
        if (s == null) return "";
        int score; String reason;
        switch (which) {
            case "LP": score=s.longPullScore; reason=s.longPullReason; break;
            case "LB": score=s.longBreakScore; reason=s.longBreakReason; break;
            case "SR": score=s.shortResScore; reason=s.shortResReason; break;
            default: score=s.shortBreakScore; reason=s.shortBreakReason; break;
        }
        String grade = score >= 86 ? "GÜÇLÜ SİNYAL" : score >= 78 ? "SİNYAL" : score >= 68 ? "TAKİP" : "İŞLEM YOK";
        return "\n\nAUTO PUAN: " + score + "/100 • " + grade + "\n" + reason;
    }

    private String formatTime(long millis) {
        java.text.SimpleDateFormat f = new java.text.SimpleDateFormat("dd.MM HH:mm", Locale.getDefault());
        return f.format(new java.util.Date(millis));
    }

    private void toggleAutoMode() {
        android.content.SharedPreferences p = getSharedPreferences(MonitorService.PREFS, MODE_PRIVATE);
        boolean next = !p.getBoolean(MonitorService.KEY_AUTO_MODE, true);
        p.edit().putBoolean(MonitorService.KEY_AUTO_MODE, next).apply();
        Toast.makeText(this, next ? "AUTO HIGH ACCURACY açıldı. Yeni 15m kapanışta derin analiz yapılacak." : "AUTO kapatıldı; manuel plan seviyeleri kullanılacak.", Toast.LENGTH_LONG).show();
        setContentView(buildUi());
        refreshStatus();
    }

        input.setGravity(Gravity.TOP);

                .setTitle("Plan kodlarını yapıştır")

                .create();

                    .replace("```text", "")
                    .replace("```", "")
                    .trim();
            if (rawText.isEmpty()) {
                Toast.makeText(this, "Plan kodu boş.", Toast.LENGTH_LONG).show();
                return;
            }

            String[] lines = rawText.split("\\r?\\n");

                    String[] a = code.replace(',', '.').split("\\|");
